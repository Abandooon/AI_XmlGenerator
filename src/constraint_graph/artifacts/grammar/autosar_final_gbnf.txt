# AUTOSAR Final GBNF Grammar
# 基于完整ASW_COM.arxml，解决空格问题，支持多端口和完整结构

start: autosar_component

autosar_component: "<APPLICATION-SW-COMPONENT-TYPE>" component_content "</APPLICATION-SW-COMPONENT-TYPE>"

component_content: short_name admin_data? ports internal_behaviors

# 基础元素
short_name: "<SHORT-NAME>" name_value "</SHORT-NAME>"
name_value: /[A-Za-z][A-Za-z0-9_]*/

# 可选的管理数据
admin_data: "<ADMIN-DATA>" "<SDGS>" "<SDG" " " "GID=\"Master\">" "<SD>true</SD>" "</SDG>" "</SDGS>" "</ADMIN-DATA>"

# PORTS - 支持多个P-PORT和R-PORT
ports: "<PORTS>" port_list "</PORTS>"
port_list: port_item port_item* 
port_item: p_port | r_port

# P-PORT-PROTOTYPE (多个提供端口)
p_port: "<P-PORT-PROTOTYPE" " " uuid_attr ">" p_port_content "</P-PORT-PROTOTYPE>"
p_port_content: short_name provided_interface_tref

provided_interface_tref: "<PROVIDED-INTERFACE-TREF" " " "DEST=\"SENDER-RECEIVER-INTERFACE\">" interface_path "</PROVIDED-INTERFACE-TREF>"

# R-PORT-PROTOTYPE (多个需求端口，包含COM-SPECS)
r_port: "<R-PORT-PROTOTYPE" " " uuid_attr ">" r_port_content "</R-PORT-PROTOTYPE>"
r_port_content: short_name required_com_specs required_interface_tref

required_com_specs: "<REQUIRED-COM-SPECS>" nonqueued_receiver_com_spec "</REQUIRED-COM-SPECS>"
nonqueued_receiver_com_spec: "<NONQUEUED-RECEIVER-COM-SPEC>" com_spec_content "</NONQUEUED-RECEIVER-COM-SPEC>"
com_spec_content: data_element_ref alive_timeout handle_timeout_type

data_element_ref: "<DATA-ELEMENT-REF" " " "DEST=\"VARIABLE-DATA-PROTOTYPE\">" data_element_path "</DATA-ELEMENT-REF>"
alive_timeout: "<ALIVE-TIMEOUT>" timeout_value "</ALIVE-TIMEOUT>"
handle_timeout_type: "<HANDLE-TIMEOUT-TYPE>NONE</HANDLE-TIMEOUT-TYPE>"

required_interface_tref: "<REQUIRED-INTERFACE-TREF" " " "DEST=\"SENDER-RECEIVER-INTERFACE\">" interface_path "</REQUIRED-INTERFACE-TREF>"

# INTERNAL-BEHAVIORS
internal_behaviors: "<INTERNAL-BEHAVIORS>" swc_internal_behavior "</INTERNAL-BEHAVIORS>"
swc_internal_behavior: "<SWC-INTERNAL-BEHAVIOR>" behavior_content "</SWC-INTERNAL-BEHAVIOR>"
behavior_content: short_name events runnables

# EVENTS
events: "<EVENTS>" timing_event "</EVENTS>"
timing_event: "<TIMING-EVENT" " " uuid_attr ">" timing_event_content "</TIMING-EVENT>"
timing_event_content: short_name start_on_event_ref period

start_on_event_ref: "<START-ON-EVENT-REF" " " "DEST=\"RUNNABLE-ENTITY\">" runnable_ref_path "</START-ON-EVENT-REF>"
period: "<PERIOD>" period_value "</PERIOD>"

# RUNNABLES (包含完整的数据访问点)
runnables: "<RUNNABLES>" runnable_entity "</RUNNABLES>"
runnable_entity: "<RUNNABLE-ENTITY" " " uuid_attr ">" runnable_content "</RUNNABLE-ENTITY>"
runnable_content: short_name data_receive_points data_send_points symbol

# DATA-RECEIVE-POINT-BY-ARGUMENTS (多个变量访问)
data_receive_points: "<DATA-RECEIVE-POINT-BY-ARGUMENTS>" variable_access_list "</DATA-RECEIVE-POINT-BY-ARGUMENTS>"

# DATA-SEND-POINTS (多个变量访问)
data_send_points: "<DATA-SEND-POINTS>" variable_access_list "</DATA-SEND-POINTS>"

variable_access_list: variable_access variable_access variable_access | variable_access variable_access | variable_access
variable_access: "<VARIABLE-ACCESS" " " uuid_attr ">" variable_access_content "</VARIABLE-ACCESS>"
variable_access_content: short_name accessed_variable

accessed_variable: "<ACCESSED-VARIABLE>" autosar_variable_iref "</ACCESSED-VARIABLE>"
autosar_variable_iref: "<AUTOSAR-VARIABLE-IREF>" port_prototype_ref target_data_prototype_ref "</AUTOSAR-VARIABLE-IREF>"

port_prototype_ref: "<PORT-PROTOTYPE-REF" " " port_dest_attr ">" port_ref_path "</PORT-PROTOTYPE-REF>"
target_data_prototype_ref: "<TARGET-DATA-PROTOTYPE-REF" " " "DEST=\"VARIABLE-DATA-PROTOTYPE\">" data_prototype_path "</TARGET-DATA-PROTOTYPE-REF>"

symbol: "<SYMBOL>" symbol_name "</SYMBOL>"

# 属性定义
uuid_attr: "UUID=\"" uuid_value "\""
uuid_value: /[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12}/

port_dest_attr: "DEST=\"R-PORT-PROTOTYPE\"" | "DEST=\"P-PORT-PROTOTYPE\""

# 值定义
timeout_value: /[0-9]+\.[0-9]+/
period_value: /[0-9]+\.[0-9]+/
symbol_name: /[A-Za-z_][A-Za-z0-9_]*/

# 路径定义 (基于实际XML中的路径模式)
interface_path: "/COM_Interface/" interface_name
interface_name: "SR_Interface_MCU01_EmergShutDown" | "SR_Interface_MCU02_MaxTor" | "SR_Interface_MCU03_NRF_IdcSamp" | "SR_Interface_HCU01_TqCmd" | "SR_Interface_HCU01_Shift" | "SR_Interface_HCU02_Poweroff"

data_element_path: "/COM_Interface/" data_element_interface_path
data_element_interface_path: "SR_Interface_HCU01_TqCmd/HCU01_Tq_Cmd" | "SR_Interface_HCU01_Shift/HCU01_Shift" | "SR_Interface_HCU02_Poweroff/HCU02_Poweroff"

runnable_ref_path: "/COM_SWC/ASW_COM/SwcInternalBehavior/RE_COM_SWC"

port_ref_path: "/COM_SWC/ASW_COM/" port_ref_name
port_ref_name: "RPort_HCU01_Shift" | "RPort_HCU01_TqCmd" | "RPort_HCU02_Poweroff" | "PPort_MCU03_NRF_IdcSamp" | "PPort_MCU01_EmergShutDown" | "PPort_MCU02_MaxTor"

data_prototype_path: "/COM_Interface/" data_prototype_interface_path
data_prototype_interface_path: "SR_Interface_HCU01_Shift/HCU01_Shift" | "SR_Interface_HCU01_TqCmd/HCU01_Tq_Cmd" | "SR_Interface_HCU02_Poweroff/HCU02_Poweroff" | "SR_Interface_MCU03_NRF_IdcSamp/MCU03_NRF_IdcSamp" | "SR_Interface_MCU01_EmergShutDown/MCU01_EmergShutDown" | "SR_Interface_MCU02_MaxTor/MCU02_MaxTor"
