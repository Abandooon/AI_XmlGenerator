# AUTOSAR Practical GBNF Grammar
# 实用版本 - 重点解决空格问题，支持多端口，接近目标XML

start: autosar_component

autosar_component: "<APPLICATION-SW-COMPONENT-TYPE>" component_content "</APPLICATION-SW-COMPONENT-TYPE>"

component_content: short_name ports internal_behaviors

# 基础元素
short_name: "<SHORT-NAME>" name_value "</SHORT-NAME>"
name_value: /[A-Za-z][A-Za-z0-9_]*/

# PORTS - 支持3个P-PORT和3个R-PORT (匹配目标XML)
ports: "<PORTS>" three_p_ports three_r_ports "</PORTS>"

three_p_ports: p_port p_port p_port
three_r_ports: r_port r_port r_port

# P-PORT-PROTOTYPE (确保空格正确)
p_port: "<P-PORT-PROTOTYPE" " " uuid_attr ">" p_port_content "</P-PORT-PROTOTYPE>"
p_port_content: short_name provided_interface_tref

provided_interface_tref: "<PROVIDED-INTERFACE-TREF" " " "DEST=\"SENDER-RECEIVER-INTERFACE\">" p_interface_path "</PROVIDED-INTERFACE-TREF>"

# R-PORT-PROTOTYPE (包含COM-SPECS，确保空格正确)
r_port: "<R-PORT-PROTOTYPE" " " uuid_attr ">" r_port_content "</R-PORT-PROTOTYPE>"
r_port_content: short_name required_com_specs required_interface_tref

required_com_specs: "<REQUIRED-COM-SPECS>" nonqueued_receiver_com_spec "</REQUIRED-COM-SPECS>"
nonqueued_receiver_com_spec: "<NONQUEUED-RECEIVER-COM-SPEC>" com_spec_content "</NONQUEUED-RECEIVER-COM-SPEC>"
com_spec_content: data_element_ref alive_timeout handle_timeout_type

data_element_ref: "<DATA-ELEMENT-REF" " " "DEST=\"VARIABLE-DATA-PROTOTYPE\">" r_data_element_path "</DATA-ELEMENT-REF>"
alive_timeout: "<ALIVE-TIMEOUT>0.3</ALIVE-TIMEOUT>"
handle_timeout_type: "<HANDLE-TIMEOUT-TYPE>NONE</HANDLE-TIMEOUT-TYPE>"

required_interface_tref: "<REQUIRED-INTERFACE-TREF" " " "DEST=\"SENDER-RECEIVER-INTERFACE\">" r_interface_path "</REQUIRED-INTERFACE-TREF>"

# INTERNAL-BEHAVIORS
internal_behaviors: "<INTERNAL-BEHAVIORS>" swc_internal_behavior "</INTERNAL-BEHAVIORS>"
swc_internal_behavior: "<SWC-INTERNAL-BEHAVIOR>" behavior_content "</SWC-INTERNAL-BEHAVIOR>"
behavior_content: short_name events runnables

# EVENTS
events: "<EVENTS>" timing_event "</EVENTS>"
timing_event: "<TIMING-EVENT" " " uuid_attr ">" timing_event_content "</TIMING-EVENT>"
timing_event_content: short_name start_on_event_ref period

start_on_event_ref: "<START-ON-EVENT-REF" " " "DEST=\"RUNNABLE-ENTITY\">/COM_SWC/ASW_COM/SwcInternalBehavior/RE_COM_SWC</START-ON-EVENT-REF>"
period: "<PERIOD>0.01</PERIOD>"

# RUNNABLES
runnables: "<RUNNABLES>" runnable_entity "</RUNNABLES>"
runnable_entity: "<RUNNABLE-ENTITY" " " uuid_attr ">" runnable_content "</RUNNABLE-ENTITY>"
runnable_content: short_name data_receive_points data_send_points symbol

# 数据访问点 (简化但完整)
data_receive_points: "<DATA-RECEIVE-POINT-BY-ARGUMENTS>" three_receive_access "</DATA-RECEIVE-POINT-BY-ARGUMENTS>"
data_send_points: "<DATA-SEND-POINTS>" three_send_access "</DATA-SEND-POINTS>"

three_receive_access: variable_access variable_access variable_access
three_send_access: variable_access variable_access variable_access

variable_access: "<VARIABLE-ACCESS" " " uuid_attr ">" variable_access_content "</VARIABLE-ACCESS>"
variable_access_content: short_name accessed_variable

accessed_variable: "<ACCESSED-VARIABLE>" autosar_variable_iref "</ACCESSED-VARIABLE>"
autosar_variable_iref: "<AUTOSAR-VARIABLE-IREF>" port_prototype_ref target_data_prototype_ref "</AUTOSAR-VARIABLE-IREF>"

port_prototype_ref: "<PORT-PROTOTYPE-REF" " " port_dest_attr ">" port_ref_path "</PORT-PROTOTYPE-REF>"
target_data_prototype_ref: "<TARGET-DATA-PROTOTYPE-REF" " " "DEST=\"VARIABLE-DATA-PROTOTYPE\">" data_prototype_path "</TARGET-DATA-PROTOTYPE-REF>"

symbol: "<SYMBOL>" symbol_name "</SYMBOL>"

# 属性
uuid_attr: "UUID=\"" uuid_value "\""
uuid_value: /[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12}/

port_dest_attr: "DEST=\"R-PORT-PROTOTYPE\"" | "DEST=\"P-PORT-PROTOTYPE\""
symbol_name: /[A-Za-z_][A-Za-z0-9_]*/

# 路径模式 (基于实际XML)
p_interface_path: "/COM_Interface/SR_Interface_MCU01_EmergShutDown" | "/COM_Interface/SR_Interface_MCU02_MaxTor" | "/COM_Interface/SR_Interface_MCU03_NRF_IdcSamp"

r_interface_path: "/COM_Interface/SR_Interface_HCU01_TqCmd" | "/COM_Interface/SR_Interface_HCU01_Shift" | "/COM_Interface/SR_Interface_HCU02_Poweroff"

r_data_element_path: "/COM_Interface/SR_Interface_HCU01_TqCmd/HCU01_Tq_Cmd" | "/COM_Interface/SR_Interface_HCU01_Shift/HCU01_Shift" | "/COM_Interface/SR_Interface_HCU02_Poweroff/HCU02_Poweroff"

port_ref_path: "/COM_SWC/ASW_COM/RPort_HCU01_Shift" | "/COM_SWC/ASW_COM/RPort_HCU01_TqCmd" | "/COM_SWC/ASW_COM/RPort_HCU02_Poweroff" | "/COM_SWC/ASW_COM/PPort_MCU03_NRF_IdcSamp" | "/COM_SWC/ASW_COM/PPort_MCU01_EmergShutDown" | "/COM_SWC/ASW_COM/PPort_MCU02_MaxTor"

data_prototype_path: "/COM_Interface/SR_Interface_HCU01_Shift/HCU01_Shift" | "/COM_Interface/SR_Interface_HCU01_TqCmd/HCU01_Tq_Cmd" | "/COM_Interface/SR_Interface_HCU02_Poweroff/HCU02_Poweroff" | "/COM_Interface/SR_Interface_MCU03_NRF_IdcSamp/MCU03_NRF_IdcSamp" | "/COM_Interface/SR_Interface_MCU01_EmergShutDown/MCU01_EmergShutDown" | "/COM_Interface/SR_Interface_MCU02_MaxTor/MCU02_MaxTor"
